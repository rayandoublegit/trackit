# Bulk Seed (~1500 creators) Implementation Plan

> **For agentic workers:** execute inline (single operational script reusing already-tested pure functions). Steps use checkbox (`- [ ]`) syntax.

**Goal:** Populate prod `creators_index` with ~1500 enriched + classified TikTok creators across all niches/tiers/countries/languages, so discovery/feed serve from the DB (no per-request API calls).

**Architecture:** A standalone resumable Node script (`scripts/seed-creators-bulk.ts`) reuses the existing `scrapecreators` + `creator-enrichment` + `creator-classify` libs, loops over all niche targets (+ localized terms), enriches+classifies each candidate, and upserts to prod Supabase via the service_role key. Run locally (no serverless limit).

**Tech Stack:** Node + `tsx`, `@supabase/supabase-js`, ScrapeCreators (Pro), Anthropic Haiku.

## Task 2: Write the bulk seed script

**Files:** Create `scripts/seed-creators-bulk.ts`

- [ ] **Step 1: Write the script**

```ts
import "dotenv/config";
import { createClient } from "@supabase/supabase-js";
import { buildSeedTargets } from "@/lib/niche-tree";
import {
  searchTikTokUsersRaw,
  fetchTikTokProfileRaw,
  fetchTikTokVideosRaw,
  parseProfile,
  parseVideos,
  extractCaptions,
} from "@/lib/scrapecreators";
import { buildEnrichmentRow } from "@/lib/creator-enrichment";
import { classifyCreator } from "@/lib/creator-classify";

const SEED_TARGET = Number(process.env.SEED_TARGET ?? 1500);
const PER_NICHE = Number(process.env.SEED_PER_NICHE ?? 8);
const CONCURRENCY = Number(process.env.SEED_CONCURRENCY ?? 5);
const MIN_FOLLOWERS = 5000;

const supa = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!,
  { auth: { persistSession: false } }
);

// Localized niche keywords -> language/country diversity (FR/ES/DE/PT/IT).
const LOCALIZED = [
  "musculation", "perte de poids", "recettes", "maquillage", "mode femme", "cuisine rapide", "finance perso",
  "fitness mujeres", "maquillaje", "recetas", "moda", "finanzas personales", "perder peso",
  "fitness frauen", "abnehmen", "rezepte", "schminken", "mode",
  "emagrecer", "receitas", "maquiagem", "moda masculina",
  "ricette", "trucco", "palestra",
];

type SCUser = {
  unique_id?: string; nickname?: string; follower_count?: number; signature?: string;
  avatar_medium?: { url_list?: string[] }; avatar_168x168?: { url_list?: string[] };
};

async function search(query: string): Promise<SCUser[]> {
  const data = (await searchTikTokUsersRaw(query)) as { user_list?: { user_info?: SCUser }[] };
  return (data?.user_list ?? []).map((u) => u.user_info).filter((u): u is SCUser => !!u && !!u.unique_id);
}

async function enrichAndStore(u: SCUser, niche: string, seen: Set<string>): Promise<"ok" | "skip" | "fail"> {
  const username = String(u.unique_id);
  if (seen.has(username)) return "skip";
  seen.add(username);
  try {
    const [pRaw, vRaw] = await Promise.all([fetchTikTokProfileRaw(username), fetchTikTokVideosRaw(username)]);
    const profile = parseProfile(pRaw);
    const followers = profile.followers || Number(u.follower_count || 0);
    if (followers < MIN_FOLLOWERS) return "skip";
    const videos = parseVideos(vRaw);
    const row = buildEnrichmentRow(username, { ...profile, followers }, videos);
    let extra: Record<string, unknown> = { primary_niche: niche, niches: [niche] };
    try {
      const c = await classifyCreator({ displayName: profile.displayName, bio: profile.bio, captions: extractCaptions(vRaw) });
      extra = {
        primary_niche: c.primaryNiche,
        niches: Array.from(new Set([...c.niches, c.primaryNiche, niche])),
        language: c.language, country_code: c.countryCode, email: c.email,
      };
    } catch { /* classification best-effort */ }
    const avatar =
      u.avatar_medium?.url_list?.[0] || u.avatar_168x168?.url_list?.[0] ||
      `https://ui-avatars.com/api/?name=${encodeURIComponent(profile.displayName || username)}&background=e5e5e5&color=9a9a9a&size=200&bold=true&rounded=true`;
    await supa.from("creators_index").upsert({ ...row, avatar_url: avatar, ...extra }, { onConflict: "username" });
    return "ok";
  } catch {
    return "fail";
  }
}

async function main() {
  const seen = new Set<string>();
  // Resume: skip creators already enriched.
  const { data: existing } = await supa.from("creators_index").select("username").eq("enrichment_status", "enriched").limit(20000);
  for (const r of existing ?? []) seen.add((r as { username: string }).username);
  let stored = seen.size;
  console.log(`Resume: ${stored} already enriched. Target ${SEED_TARGET}.`);

  const targets = [...buildSeedTargets().map((t) => t.query), ...LOCALIZED];
  for (const niche of targets) {
    if (stored >= SEED_TARGET) break;
    let users: SCUser[] = [];
    try { users = await search(niche); } catch { console.log(`search failed: ${niche}`); continue; }
    const top = users.sort((a, b) => Number(b.follower_count || 0) - Number(a.follower_count || 0)).slice(0, PER_NICHE);
    for (let i = 0; i < top.length && stored < SEED_TARGET; i += CONCURRENCY) {
      const chunk = top.slice(i, i + CONCURRENCY);
      const res = await Promise.all(chunk.map((u) => enrichAndStore(u, niche, seen)));
      stored += res.filter((r) => r === "ok").length;
    }
    console.log(`[${niche}] stored=${stored}/${SEED_TARGET}`);
    await new Promise((r) => setTimeout(r, 250));
  }
  console.log(`DONE. ~${stored} creators in creators_index.`);
  process.exit(0);
}

main().catch((e) => { console.error(e); process.exit(1); });
```
